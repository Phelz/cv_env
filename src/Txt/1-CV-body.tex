\section{Research Contributions}

\begin{Summary}{HAICU Trap Commissioning -- 2025--\textit{(present)}, Ph.D. Thesis}

	\veritem{Built various components of the magnetic trap.}\\

	\veritem{Tested the magnets with running cooling water and current up to 1200 Amps.}\\

	\veritem{Researching and developing strategies for integrating temperature sensors into the magnetic trap for monitoring and interlocking the magnets during operation.}

\end{Summary}


\begin{Summary}{HAICU Experiment Simulations -- 2024--\textit{(present)}, Ph.D. Thesis}

	\veritem{Developed COMSOL Multiphysics simulations in conjuction with C++ code to model the magnetic field used in the HAICU experiment.}\\

	\veritem{Integrated the field maps into a C++ software package to simulate the trajectories of hydrogen in our apparatus.}\\

	\veritem{Used the simulations to time the trapping of hydrogen, optimizing for maximum number of atoms trapped.}\\

	\veritem{Extended the simulations to include magnetic deceleration of hydrogen.}

\end{Summary}

\begin{Summary}{Radio Communication Disruption Forecasting -- 2023--2024, M.Sc. Thesis}
	\veritem{Performed an ordinal statistical analysis of over 4 years of radio and riometer data.}\\

	% \veritem{The data were filtered for space weather conditions such as solar flares and geomagnetic storms/substorms, as well as seasonal variations.}\\

	\veritem{By finding correlations between the two datasets, I established the riometer response as proxy for high-frequency radio communication disruption.}\\

	\veritem{Identified the core drivers for signal loss for a wide range of frequencies from 5 to 15 MHz.}\\

	\veritem{Findings currently under review for publication in the \textit{Journal of Geophysical Research: Space Physics.}}
\end{Summary}


\begin{Summary}{Sizing Radio-disrupted Regions -- 2022--2023, M.Sc. Thesis}
	\veritem{Wrote a ray tracing program to model the undisturbed propagation of radio waves.}\\

	\veritem{Used the modelled signal, together with the signal status observed by the receiver, to estimate the size of the region where radio waves are disrupted.}\\

	\veritem{Extended the methodology to a network of multiple radio transmitters and receivers across Canada.}\\

	\veritem{The technique was published in: \textit{Journal of Geophysical Research: Space Physics}: \href{https://onlinelibrary.wiley.com/doi/abs/10.1029/2023JA032375}{Ghaly et al., 2024.}}All code, data analysis, visualizations and writing were done by me. Coauthors helped with editing the paper.
\end{Summary}

\section{Collaboration and Impact}

\begin{Summary}{Natural Resources Canada (NRCan) Collaboration -- 2020--2024 Research Assistant and M.Sc. Thesis}
	\veritem{Upon discovering NRCan's underutilized time-signal transmitters, I verified their dectability in our data.}\\

	\veritem{I then established that their signals were disrupted by space weather, and collaborated with NRCan to incorporate their network of radio transmitters into my analysis.}\\

	\veritem{The success of this collaboration led to the development of a space weather forecasting/alerting system for modeling where and when radio communications are disrupted.}


\end{Summary}

\section{Software and Technical Development}

\begin{Summary}{MITACS Globalink Award: CERN Control System Revamp -- Summer 2025}
	\veritem{I was awarded a MITACS Globalink Research Award to spend the summer of 2025 at CERN, revamping the control system for the microwaves in the ALPHA experiment.}\\

	\veritem{Upgrading from a National Instruments PXI crate to a compact-RIO, I developed entirely new control software in LabVIEW.}\\

	\veritem{The new software uses the Remote Action Engine framework, allowing for remote control and monitoring of the microwave system.}\\

\end{Summary}

\begin{Summary}{Real-time Data Pipeline and Automatic Baselining -- 2021--2023, Research Assistant and M.Sc. Thesis}

	\veritem{Adapted the Perseus, Software-Defined Radio (SDR), making it usable for research.}\\

	\veritem{Wrote software for data acquisition in Python and Java and tested the instrument's ability to function as a receiver.}\\

	\veritem{Developed an online data browser for the prototype instruments deployed to generate curated, raw (K0-level) data for the public. Back-processed and published K0 data for 2022 and 2023.}\\

	\veritem{Wrote instrument-side code to generate curated K0 data on-site for all instruments deployed. Data are now being published in real-time.}\\

	\veritem{Developed a Graphical User Interface (GUI) for semi-automatic, instrument-specific baselining.}\\

	\veritem{Developed a methodology for real-time, automatic baselining to generate operational (K2-level) data.}\\

	\veritem{Designed a web application with a Postgre database that browses through different specification parameters for flagging space weather events.}

\end{Summary}
